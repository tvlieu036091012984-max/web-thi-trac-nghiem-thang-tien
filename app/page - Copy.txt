'use client'

import { useState, useEffect } from 'react'
import { createClient } from '@/utils/supabase/client'
import { GraduationCap, Sparkles, Trophy, ArrowRight, BookOpen, Calendar, Clock, ShieldCheck, Loader2 } from 'lucide-react'
import Link from 'next/link'

export default function Home() {
  const [exams, setExams] = useState<any[]>([])
  const [leaderboard, setLeaderboard] = useState<any[]>([])
  const [loading, setLoading] = useState(true)
  const [selectedExamId, setSelectedExamId] = useState<string>('')

  useEffect(() => {
    fetchHomeData()
  }, [])

  const fetchHomeData = async () => {
    try {
      const supabase = createClient()
      
      // Lấy danh sách đề thi
      const { data: examData } = await supabase.from('exams').select('*').order('created_at', { ascending: false })
      if (examData && examData.length > 0) {
        setExams(examData)
        setSelectedExamId(examData[0].id)
        fetchLeaderboard(examData[0].id)
      }
      setLoading(false)
    } catch (err) {
      console.error("Lỗi tải trang chủ:", err)
      setLoading(false)
    }
  }

  const fetchLeaderboard = async (examId: string) => {
    try {
      const supabase = createClient()
      const { data } = await supabase
        .from('submissions')
        .select('*')
        .eq('exam_id', examId)
        .order('score', { ascending: false })
        .order('created_at', { ascending: true })
        .limit(10)
      if (data) setLeaderboard(data)
    } catch (err) {
      console.error("Lỗi tải bảng xếp hạng:", err)
    }
  }

  const handleExamChange = (e: React.ChangeEvent<HTMLSelectElement>) => {
    const id = e.target.value
    setSelectedExamId(id)
    fetchLeaderboard(id)
  }
if (loading) return (
    <div className="min-h-screen flex items-center justify-center bg-slate-50">
      <Loader2 className="w-10 h-10 animate-spin text-blue-600" />
    </div>
  )

  return (
    <div className="min-h-screen bg-slate-50 font-sans flex flex-col justify-between">
      <div className="max-w-6xl mx-auto px-4 py-8 md:py-12 w-full">
        
        {/* BANNER CHÀO MỪNG */}
        <div className="bg-white p-8 md:p-12 rounded-3xl shadow-sm border border-slate-200 text-center mb-10 relative overflow-hidden">
          <div className="w-20 h-20 bg-blue-600 rounded-3xl flex items-center justify-center mx-auto mb-6 shadow-lg shadow-blue-200">
            <GraduationCap className="text-white w-10 h-10" />
          </div>

          <div className="inline-flex items-center space-x-2 bg-orange-100 text-orange-700 px-4 py-1.5 rounded-full text-xs font-bold mb-4">
            <Sparkles className="w-4 h-4" /> <span>Hệ Thống Kiểm Tra Đánh Giá Trực Tuyến</span>
          </div>

          <h1 className="text-3xl md:text-5xl font-black text-slate-800 tracking-tight mb-3">
            LỚP HỌC THĂNG TIẾN
          </h1>
          <p className="text-blue-600 font-bold text-base md:text-lg mb-4 flex items-center justify-center">
            <ShieldCheck className="w-5 h-5 mr-1.5" /> Giáo viên phụ trách: Thầy TRẦN VĂN LIỆU
          </p>
          <p className="text-slate-500 text-sm max-w-xl mx-auto mb-8">
            Chọn đề thi bên dưới để bắt đầu làm bài hoặc theo dõi bảng vàng vinh danh các em học sinh đạt thành tích xuất sắc!
          </p>

          <div className="flex justify-center gap-4">
            <Link 
              href="/admin/dashboard" 
              className="bg-blue-600 hover:bg-blue-700 text-white text-sm font-bold px-6 py-3.5 rounded-2xl shadow-md transition flex items-center active:scale-95"
            >
              Quản Trị Dashboard ⚙️
            </Link>
          </div>
        </div>

        {/* NỘI DUNG CHÍNH: 2 CỘT (DANH SÁCH ĐỀ & BẢNG XẾP HẠNG) */}
        <div className="grid grid-cols-1 md:grid-cols-3 gap-8">
          
          {/* CỘT TRÁI: DANH SÁCH ĐỀ THI CHO HỌC SINH */}
          <div className="md:col-span-2 space-y-4">
            <h2 className="text-lg font-black text-slate-800 flex items-center mb-4">
              <BookOpen className="w-5 h-5 mr-2 text-blue-600" /> Danh Sách Đề Thi Mới Nhất ({exams.length})
            </h2>

            {exams.map((exam) => (
              <div key={exam.id} className="bg-white p-6 rounded-2xl shadow-sm border border-slate-200 hover:shadow-md transition flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4">
                <div>
                  <div className="flex items-center space-x-2 mb-2">
                    <span className="bg-blue-50 text-blue-700 text-xs font-bold px-3 py-1 rounded-full flex items-center">
                      <Clock className="w-3 h-3 mr-1" /> Thời gian: {exam.config?.timeLimit || 40} phút
                    </span>
                    <span className="text-slate-400 text-xs flex items-center">
                      <Calendar className="w-3 h-3 mr-1" /> {new Date(exam.created_at).toLocaleDateString('vi-VN')}
                    </span>
                  </div>
                  <h3 className="text-base font-bold text-slate-800">{exam.title}</h3>
                </div>

                <Link 
                  href={`/exam/${exam.id}`}
                  className="w-full sm:w-auto bg-orange-500 hover:bg-orange-600 text-white text-xs font-bold px-5 py-3 rounded-xl shadow-sm flex items-center justify-center transition active:scale-95 flex-shrink-0"
                >
                  Vào Làm Bài <ArrowRight className="w-4 h-4 ml-1.5" />
                </Link>
              </div>
            ))}

            {exams.length === 0 && (
              <div className="text-center py-16 bg-white rounded-2xl border border-dashed border-slate-300 text-slate-400">
                Chưa có đề thi nào được xuất bản. Thầy hãy tạo và đăng đề từ trang quản trị nhé!
              </div>
            )}
          </div>

          {/* CỘT PHẢI: BẢNG XẾP HẠNG (LEADERBOARD) */}
          <div className="bg-white p-6 rounded-2xl shadow-sm border border-slate-200 h-fit">
            <h2 className="text-base font-black text-slate-800 flex items-center mb-4">
              <Trophy className="w-5 h-5 mr-2 text-amber-500" /> Bảng Vàng Vinh Danh
            </h2>

            {exams.length > 0 && (
              <div className="mb-4">
                <label className="block text-xs font-bold text-slate-500 mb-1.5">Chọn đề xem bảng xếp hạng:</label>
                <select 
                  value={selectedExamId}
                  onChange={handleExamChange}
                  className="w-full p-2.5 bg-slate-50 border border-slate-200 rounded-xl text-xs font-bold text-slate-800 outline-none focus:border-blue-500"
                >
                  {exams.map(ex => (
                    <option key={ex.id} value={ex.id}>{ex.title}</option>
                  ))}
                </select>
              </div>
            )}

            <div className="space-y-3">
              {leaderboard.map((item, index) => (
                <div key={item.id} className={`p-3.5 rounded-2xl border flex items-center justify-between ${index === 0 ? 'bg-amber-50/80 border-amber-300' : index === 1 ? 'bg-slate-100 border-slate-300' : index === 2 ? 'bg-orange-50/60 border-orange-200' : 'bg-slate-50/50 border-slate-100'}`}>
                  <div className="flex items-center space-x-3 truncate">
                    <span className={`w-6 h-6 rounded-xl flex items-center justify-center text-xs font-black ${index === 0 ? 'bg-amber-500 text-white' : index === 1 ? 'bg-slate-400 text-white' : index === 2 ? 'bg-orange-400 text-white' : 'bg-slate-200 text-slate-600'}`}>
                      {index + 1}
                    </span>
                    <span className="text-xs font-bold text-slate-800 truncate max-w-[120px]">{item.student_name}</span>
                  </div>
                  <span className="text-xs font-black text-blue-600 bg-white px-2.5 py-1 rounded-xl shadow-sm">
                    {item.score} đ
                  </span>
                </div>
              ))}

              {leaderboard.length === 0 && (
                <div className="text-center py-12 text-slate-400 text-xs">
                  Chưa có học sinh nộp bài cho đề này.
                </div>
              )}
            </div>

          </div>

        </div>

      </div>

      <footer className="text-center py-6 text-xs text-slate-400 border-t border-slate-200 bg-white mt-12">
        Hệ thống học tập thông minh • Phát triển bởi <strong className="text-slate-600">Thầy Trần Văn Liệu</strong> • <span className="text-blue-600 font-bold">Lớp Học Thêm Thăng Tiến</span>
      </footer>
    </div>
  )
}
